    <script>
        // ==========================================
        // CONFIGURATION & ETAT GLOBAL
        // ==========================================
        const GITHUB_JSON_URL = 'https://raw.githubusercontent.com/Educsmart/educsmartdoc/refs/heads/main/documents.json';
        const state = {
            documents: [],
            filteredDocs: [],
            filters: {
                search: '',
                subject: [],
                level: []
            },
            sortBy: 'recent'
        };

        // ==========================================
        // INITIALISATION
        // ==========================================
        document.addEventListener('DOMContentLoaded', () => {
            initTheme();
            setupEventListeners();
            fetchDocuments();
            window.addEventListener('hashchange', handleRouting);
        });

        // ==========================================
        // FETCH & DATA
        // ==========================================
        async function fetchDocuments() {
            renderSkeletons();
            try {
                const response = await fetch(GITHUB_JSON_URL);
                if (!response.ok) throw new Error('Erreur réseau');
                const data = await response.json();
                state.documents = data;
                
                extractFilters(data);
                applyFiltersAndSort();
                handleRouting(); 
            } catch (error) {
                console.error("Erreur de chargement:", error);
                document.getElementById('resultsCount').textContent = "Erreur de chargement des données.";
            } finally {
                document.getElementById('loader').classList.add('hidden');
                document.getElementById('documentsGrid').classList.remove('hidden');
            }
        }

        // ==========================================
        // LOGIQUE DE FILTRAGE ET TRI
        // ==========================================
        function applyFiltersAndSort() {
            let result = state.documents;

            if (state.filters.search) {
                const term = state.filters.search.toLowerCase();
                result = result.filter(doc => 
                    doc.title.toLowerCase().includes(term) ||
                    doc.tags.some(tag => tag.toLowerCase().includes(term)) ||
                    doc.author.toLowerCase().includes(term)
                );
            }

            if (state.filters.subject.length > 0) {
                result = result.filter(doc => state.filters.subject.includes(doc.subject));
            }

            if (state.filters.level.length > 0) {
                result = result.filter(doc => state.filters.level.includes(doc.class_level));
            }

            result.sort((a, b) => {
                if (state.sortBy === 'recent') return new Date(b.created_at) - new Date(a.created_at);
                if (state.sortBy === 'popular') return b.popularity - a.popularity;
                if (state.sortBy === 'alphabetical') return a.title.localeCompare(b.title);
                return 0;
            });

            state.filteredDocs = result;
            renderGrid();
        }

        // ==========================================
        // RENDU UI
        // ==========================================
        function renderGrid() {
            const grid = document.getElementById('documentsGrid');
            const emptyState = document.getElementById('emptyState');
            const count = document.getElementById('resultsCount');

            grid.innerHTML = '';
            count.textContent = `${state.filteredDocs.length} document${state.filteredDocs.length > 1 ? 's' : ''}`;

            if (state.filteredDocs.length === 0) {
                grid.classList.add('hidden');
                emptyState.classList.remove('hidden');
                return;
            }

            grid.classList.remove('hidden');
            emptyState.classList.add('hidden');

            state.filteredDocs.forEach(doc => {
                const card = document.createElement('div');
                card.className = 'saas-card bg-white dark:bg-darkCard rounded-xl border border-gray-100 dark:border-gray-800 overflow-hidden flex flex-col cursor-pointer';
                card.onclick = () => window.location.hash = `doc/${doc.id}`;
                
                card.innerHTML = `
                    <div class="h-40 bg-gray-200 dark:bg-gray-800 relative">
                        <img src="${getDriveDirectImageUrl(doc.thumbnail)}" alt="${doc.title}" class="w-full h-full object-cover opacity-90 hover:opacity-100 transition-opacity" loading="lazy">
                        <span class="absolute top-3 right-3 bg-white/90 dark:bg-darkBg/90 backdrop-blur text-xs font-bold px-2 py-1 rounded shadow-sm text-primary">
                            ${doc.file_type}
                        </span>
                    </div>
                    <div class="p-5 flex-1 flex flex-col">
                        <div class="flex items-center gap-2 mb-2">
                            <span class="text-xs font-medium bg-indigo-50 dark:bg-indigo-500/10 text-indigo-600 dark:text-indigo-400 px-2 py-1 rounded">
                                ${doc.subject}
                            </span>
                            <span class="text-xs text-gray-500">${doc.class_level}</span>
                        </div>
                        <h3 class="font-bold text-gray-900 dark:text-white leading-tight mb-2 line-clamp-2">${doc.title}</h3>
                        <p class="text-sm text-gray-500 dark:text-gray-400 mb-4 flex-1 line-clamp-2">${doc.description}</p>
                        
                        <div class="flex items-center justify-between border-t border-gray-100 dark:border-gray-800 pt-4 mt-auto">
                            <span class="text-xs text-gray-400"><i class="ph ph-user"></i> ${doc.author}</span>
                            <div class="flex gap-2">
                                <button onclick="event.stopPropagation(); window.open('${doc.drive_download_link}', '_blank')" class="p-1.5 text-gray-400 hover:text-primary transition-colors" title="Télécharger">
                                    <i class="ph ph-download-simple text-lg"></i>
                                </button>
                            </div>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function renderDetail(id) {
            const doc = state.documents.find(d => d.id === id);
            if (!doc) {
                window.location.hash = '';
                return;
            }

            const container = document.getElementById('detailContent');
            const shareUrl = window.location.href;

            container.innerHTML = `
                <div class="flex flex-col md:flex-row gap-8">
                    <div class="w-full md:w-1/3">
                        <img src="${getDriveDirectImageUrl(doc.thumbnail)}" alt="Cover" class="w-full rounded-lg shadow-md border border-gray-100 dark:border-gray-800">
                        <div class="mt-6 flex flex-col gap-3">
                            <a href="${doc.drive_view_link}" target="_blank" class="w-full bg-primary hover:bg-primaryHover text-white text-center font-medium py-3 rounded-lg transition-colors flex justify-center items-center gap-2">
                                <i class="ph ph-book-open"></i> Lire le document
                            </a>
                            <a href="${doc.drive_download_link}" target="_blank" class="w-full bg-gray-100 hover:bg-gray-200 dark:bg-gray-800 dark:hover:bg-gray-700 text-gray-900 dark:text-white text-center font-medium py-3 rounded-lg transition-colors flex justify-center items-center gap-2">
                                <i class="ph ph-download-simple"></i> Télécharger (${doc.file_type})
                            </a>
                            <button onclick="copyLink('${shareUrl}')" class="w-full border border-gray-200 dark:border-gray-700 hover:bg-gray-50 dark:hover:bg-gray-800 text-gray-700 dark:text-gray-300 text-center font-medium py-3 rounded-lg transition-colors flex justify-center items-center gap-2">
                                <i class="ph ph-link"></i> Partager le lien
                            </button>
                        </div>
                    </div>
                    <div class="w-full md:w-2/3">
                        <div class="flex items-center gap-3 mb-4">
                            <span class="px-3 py-1 bg-indigo-50 dark:bg-indigo-500/10 text-indigo-600 dark:text-indigo-400 rounded-full text-sm font-medium">${doc.subject}</span>
                            <span class="px-3 py-1 bg-gray-100 dark:bg-gray-800 text-gray-600 dark:text-gray-300 rounded-full text-sm font-medium">${doc.class_level}</span>
                        </div>
                        <h1 class="text-3xl md:text-4xl font-bold text-gray-900 dark:text-white mb-4">${doc.title}</h1>
                        <p class="text-lg text-gray-600 dark:text-gray-300 mb-8">${doc.description}</p>
                        
                        <h3 class="text-sm font-semibold text-gray-900 dark:text-white uppercase tracking-wider mb-4 border-b border-gray-100 dark:border-gray-800 pb-2">Informations</h3>
                        <div class="grid grid-cols-2 gap-4 text-sm">
                            <div>
                                <p class="text-gray-500">Auteur</p>
                                <p class="font-medium text-gray-900 dark:text-white">${doc.author}</p>
                            </div>
                            <div>
                                <p class="text-gray-500">Pages</p>
                                <p class="font-medium text-gray-900 dark:text-white">${doc.number_of_pages}</p>
                            </div>
                            <div>
                                <p class="text-gray-500">Publication</p>
                                <p class="font-medium text-gray-900 dark:text-white">${doc.publication_date}</p>
                            </div>
                            <div>
                                <p class="text-gray-500">Mots-clés</p>
                                <div class="flex gap-2 mt-1 flex-wrap">
                                    ${doc.tags.map(t => `<span class="text-xs bg-gray-100 dark:bg-gray-800 px-2 py-1 rounded text-gray-600 dark:text-gray-400">${t}</span>`).join('')}
                                </div>
                            </div>
                        </div>
                    </div>
                </div>
            `;
        }

        // ==========================================
        // ROUTING & NAVIGATION
        // ==========================================
        function handleRouting() {
            const hash = window.location.hash;
            const viewHome = document.getElementById('view-home');
            const viewDetail = document.getElementById('view-detail');

            if (hash.startsWith('#doc/')) {
                const id = hash.split('/')[1];
                viewHome.classList.remove('active');
                viewDetail.classList.add('active');
                window.scrollTo(0, 0);
                renderDetail(id);
            } else {
                viewDetail.classList.remove('active');
                viewHome.classList.add('active');
            }
        }

        // ==========================================
        // UTILITAIRES & EVENTS
        // ==========================================
        
        // NOUVEAU : Fonction pour convertir un lien Google Drive standard en lien d'image directe
        function getDriveDirectImageUrl(url) {
            if (!url) return '';
            // Cherche l'ID du fichier dans l'URL Drive
            const match = url.match(/\/d\/([a-zA-Z0-9_-]+)/) || url.match(/id=([a-zA-Z0-9_-]+)/);
            if (match && match[1]) {
                // Retourne un lien lisible par une balise <img>
                return `https://drive.google.com/uc?export=view&id=${match[1]}`;
            }
            return url;
        }

        function setupEventListeners() {
            const searchInput = document.getElementById('searchInput');
            let timeout = null;
            searchInput.addEventListener('input', (e) => {
                clearTimeout(timeout);
                timeout = setTimeout(() => {
                    state.filters.search = e.target.value;
                    applyFiltersAndSort();
                }, 300);
            });

            document.getElementById('sortSelect').addEventListener('change', (e) => {
                state.sortBy = e.target.value;
                applyFiltersAndSort();
            });

            document.getElementById('themeToggle').addEventListener('click', toggleTheme);
        }

        function extractFilters(data) {
            const subjects = [...new Set(data.map(d => d.subject))].sort();
            const levels = [...new Set(data.map(d => d.class_level))].sort();

            const generateCheckboxes = (items, containerId, filterType) => {
                const container = document.getElementById(containerId);
                container.innerHTML = items.map(item => `
                    <label class="flex items-center space-x-3 cursor-pointer group">
                        <input type="checkbox" value="${item}" class="form-checkbox h-4 w-4 text-primary rounded border-gray-300 dark:border-gray-600 dark:bg-gray-700 focus:ring-primary transition-colors"
                        onchange="handleFilterChange('${filterType}', '${item}', this.checked)">
                        <span class="text-sm text-gray-600 dark:text-gray-400 group-hover:text-gray-900 dark:group-hover:text-white transition-colors">${item}</span>
                    </label>
                `).join('');
            };

            generateCheckboxes(subjects, 'filter-subjects', 'subject');
            generateCheckboxes(levels, 'filter-levels', 'level');
        }

        window.handleFilterChange = (type, value, isChecked) => {
            if (isChecked) {
                state.filters[type].push(value);
            } else {
                state.filters[type] = state.filters[type].filter(v => v !== value);
            }
            applyFiltersAndSort();
        };

        function copyLink(url) {
            navigator.clipboard.writeText(url).then(() => {
                showToast("Lien copié dans le presse-papier !");
            });
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            document.getElementById('toastMsg').textContent = message;
            toast.classList.remove('translate-y-20', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 3000);
        }

        function renderSkeletons() {
            const loader = document.getElementById('loader');
            loader.innerHTML = Array(6).fill(`
                <div class="bg-white dark:bg-darkCard rounded-xl border border-gray-100 dark:border-gray-800 p-4 h-80 animate-pulse flex flex-col">
                    <div class="h-40 bg-gray-200 dark:bg-gray-700 rounded-lg mb-4"></div>
                    <div class="h-4 bg-gray-200 dark:bg-gray-700 rounded w-1/4 mb-2"></div>
                    <div class="h-6 bg-gray-200 dark:bg-gray-700 rounded w-3/4 mb-4"></div>
                    <div class="h-4 bg-gray-200 dark:bg-gray-700 rounded w-full mt-auto"></div>
                </div>
            `).join('');
        }

        function initTheme() {
            if (localStorage.theme === 'dark' || (!('theme' in localStorage) && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
                document.documentElement.classList.add('dark');
            } else {
                document.documentElement.classList.remove('dark');
            }
        }

        function toggleTheme() {
            document.documentElement.classList.toggle('dark');
            if (document.documentElement.classList.contains('dark')) {
                localStorage.theme = 'dark';
            } else {
                localStorage.theme = 'light';
            }
        }
    </script>
